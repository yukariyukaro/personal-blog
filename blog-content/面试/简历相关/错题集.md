实现一个通用的请求重试函数 retry，要求：
输入：一个返回 Promise 的异步函数 fn，以及配置对象 { retries, delay, backoff }。 
输出：返回一个新的函数，调用时会自动执行 fn，在失败时按配置自动重试，直到成功或达到最大重试次数。 
行为细则：每次重试前等待 delay 毫秒；若配置 backoff 为 true，则每次等待时间按指数或线性增长；重试期间应能传递原始参数并返回最终的 Promise 结果或最终错误
function retryWithTimeout(fn, retries, timeout) {
    return new Promise((resolve, reject) => {
        const retry = async (attempt) => {
            try {
                const result = await fn();
                resolve(result);
            } catch (error) {
                if (attempt < retries) {
                    setTimeout(() => {
                        retry(attempt + 1);
                    }, timeout);
                } else {
                    reject("Exceeded maximum retries");
                }
            }
        };
 
        retry(0);
    });
}
 
// 使用示例
const asyncFunction = async () => {
    // 模拟一个可能会失败的异步操作
    if (Math.random() < 0.8) {
        return "Success!";
    } else {
        throw new Error("Failed");
    }
};
 
retryWithTimeout(asyncFunction, 3, 1000)
    .then(result => console.log(result))
    .catch(error => console.error(error));
